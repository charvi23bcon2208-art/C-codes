5. Create a college management system in which a single parent class of college is extended with students and a teacher class(inheritance)

student -> roll number- mobile  number- email id- subjecr- code- year
teacher -> name- age- department- salary

and call a virtual function to show their classes







#include <iostream>
#include <string>
#include <vector>

using namespace std;

// Base Parent Class
class College {
public:
    // Virtual function to be overridden by derived classes
    virtual void showDetails() = 0; 
    virtual ~College() {} // Virtual destructor for proper cleanup
};

// Derived Student Class
class Student : public College {
private:
    int rollNumber;
    string mobileNumber;
    string emailId;
    string subject;
    string code;
    int year;

public:
    Student(int roll, string mob, string email, string sub, string c, int yr) 
        : rollNumber(roll), mobileNumber(mob), emailId(email), subject(sub), code(c), year(yr) {}

    void showDetails() override {
        cout << "--- Student Details ---" << endl;
        cout << "Roll No: " << rollNumber << "\nMobile: " << mobileNumber 
             << "\nEmail: " << emailId << "\nSubject: " << subject 
             << "\nCode: " << code << "\nYear: " << year << endl << endl;
    }
};

// Derived Teacher Class
class Teacher : public College {
private:
    string name;
    int age;
    string department;
    double salary;

public:
    Teacher(string n, int a, string dept, double sal) 
        : name(n), age(a), department(dept), salary(sal) {}

    void showDetails() override {
        cout << "--- Teacher Details ---" << endl;
        cout << "Name: " << name << "\nAge: " << age 
             << "\nDepartment: " << department << "\nSalary: $" << salary << endl << endl;
    }
};

int main() {
    // Array of base class pointers to demonstrate polymorphism
    vector<College*> collegeMembers;

    // Adding records
    collegeMembers.push_back(new Student(101, "9876543210", "stud@edu.com", "Computer Science", "CS50", 2024));
    collegeMembers.push_back(new Teacher("Professor Oak", 55, "Biology", 75000.00));

    // Calling the virtual function for each object
    for (College* member : collegeMembers) {
        member->showDetails();
    }

    // Cleaning up allocated memory
    for (College* member : collegeMembers) {
        delete member;
    }

    return 0;
}
