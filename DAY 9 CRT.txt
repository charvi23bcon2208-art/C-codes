3. SUM OF ARRAY

#include<iostream>
using namespace std;
class sum
{
    public:
    int sum_of_array(int arr[], int n)
    {
        int sum = 0;
        for ( int i=0;i<n;i++)
         {
            sum += arr[i];
        }
        return sum;
    }
};
int main()
{
    int arr[5] = {1, 2, 3, 4, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    sum s1;
    int ans = s1.sum_of_array(arr, n);
    cout << ans << endl;
    return 0;
}
            


4. REVERSE ARRAY




#include<iostream>
using namespace std;
class reverse{
    
    public:
    void reverse_array(int arr[], int n)
    {
        int start = 0;
        int end = n-1;
        while(start<end)
        {
            int temp = arr[start];
            arr[start] = arr[end];
            arr[end] = temp;
            start++;
            end--;
        }
    
        for(int i =0;i<n;i++)
        {
            cout<<arr[i]<<" ";
        }
        cout<< endl;
        
        
    }
    
};


int main()
{
    int arr[5] = {1, 2, 3, 4, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    reverse s1;
    s1.reverse_array(arr, n);

    return 0;
            
}
